**Пример кода** – Мини-игра "Угадай число":

```cpp
#include <iostream>
#include <cstdlib>
#include <ctime>

int main() {
    srand(time(0));
    int secret = rand() % 100 + 1, guess;

    std::cout << "Угадай число от 1 до 100\n";
    do {
        std::cin >> guess;
        if (guess < secret) std::cout << "Слишком мало\n";
        else if (guess > secret) std::cout << "Слишком много\n";
    } while (guess != secret);

    std::cout << "Правильно!\n";
}
```

**Пример кода** – Шаблон очереди:

```cpp
template <typename T>
class Queue {
    std::vector<T> data;
public:
    void enqueue(T val) { data.push_back(val); }
    T dequeue() {
        if (data.empty()) throw std::out_of_range("Пусто");
        T val = data.front();
        data.erase(data.begin());
        return val;
    }
};
```

```cpp
void scanPort(int port) {
    sockaddr_in addr;
    addr.sin_family = AF_INET;
    addr.sin_port = htons(port);
    inet_pton(AF_INET, "127.0.0.1", &addr.sin_addr);

    int sock = socket(AF_INET, SOCK_STREAM, 0);
    if (connect(sock, (sockaddr*)&addr, sizeof(addr)) == 0)
        std::cout << "Открыт порт " << port << "\n";
    close(sock);
}
```

## 🧩 Набор 1: Начальный уровень

### Пример 1: Консольный TODO‑менеджер

**Пример кода – добавление задачи и просмотр:**

```cpp
#include <iostream>
#include <vector>
#include <string>

int main() {
    std::vector<std::string> tasks;
    std::string cmd, task;
    while (std::cin >> cmd) {
        if (cmd == "add") {
            std::getline(std::cin, task);
            tasks.push_back(task.substr(1));
        } else if (cmd == "list") {
            for (int i = 0; i < tasks.size(); ++i)
                std::cout << i+1 << ": " << tasks[i] << "\n";
        } else if (cmd == "quit") break;
    }
}
```

**Пример проекта:** `anshgoyal04/Todo-List` на GitHub — консольный менеджер с CRUD-функционалом ([GitHub][1], [GitHub][2])

## 🧠 Набор 3: Продвинутый уровень

### Пример: HTTP‑сервер на Boost.Asio

```cpp
#include <boost/asio.hpp>
using boost::asio::ip::tcp;
int main(){
    boost::asio::io_context ctx;
    tcp::acceptor acc(ctx, {tcp::v4(),8080});
    for(;;){
        tcp::socket sock(ctx);
        acc.accept(sock);
        std::string resp = "HTTP/1.1 200 OK\r\nContent-Length:5\r\n\r\nHello";
        boost::asio::write(sock, boost::asio::buffer(resp));
    }
}
```

Для отправной точки можно изучить примеры в официальной документации Boost.Asio.
